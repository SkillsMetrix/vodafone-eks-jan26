apiVersion: apps/v1
kind: Deployment
metadata:
  name: green-deploy
  labels:
    app: nginx-app
spec:
  replicas: 3
  template:
    metadata:
      labels:
        app: nginx-app
    spec:
      containers:
      - name: nginx-container
        image: cammey20/amar_custom_nginx
        ports:
        - containerPort: 80
  selector:
    matchLabels:
      app: nginx-app
